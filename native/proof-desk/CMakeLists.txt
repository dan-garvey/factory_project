cmake_minimum_required(VERSION 3.20)
project(FactoryProofDesk VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(WIN32)
  find_package(Qt6 6.11 REQUIRED COMPONENTS Core Gui Widgets Multimedia)
  qt_add_executable(factory-proof-desk WIN32 viewer.cpp)
  target_link_libraries(factory-proof-desk PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Multimedia)
  target_compile_options(factory-proof-desk PRIVATE /W4 /permissive- /utf-8)
  qt_add_executable(proof-media-fixture tests/media_fixture.cpp)
  target_link_libraries(proof-media-fixture PRIVATE Qt6::Gui Qt6::Multimedia)
  target_compile_options(proof-media-fixture PRIVATE /W4 /permissive- /utf-8)
  install(TARGETS factory-proof-desk RUNTIME DESTINATION .)
else()
  find_package(OpenSSL REQUIRED)
  add_library(proof proof.cpp codex_rpc.cpp)
  target_link_libraries(proof PUBLIC OpenSSL::Crypto)
  target_compile_options(proof PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_executable(factory-proof main.cpp)
  target_link_libraries(factory-proof PRIVATE proof)
  include(CTest)
  add_executable(proof-test tests/proof_test.cpp)
  target_link_libraries(proof-test PRIVATE proof)
  add_test(NAME proof-contract COMMAND proof-test)
  add_executable(proof-hooks-test tests/hooks_test.cpp)
  target_link_libraries(proof-hooks-test PRIVATE proof)
  add_test(NAME proof-hooks COMMAND proof-hooks-test $<TARGET_FILE:factory-proof>
    ${CMAKE_CURRENT_SOURCE_DIR}/../../.githooks/reference-transaction
    ${CMAKE_CURRENT_SOURCE_DIR}/../../.githooks/pre-push
    ${CMAKE_CURRENT_SOURCE_DIR}/../../scripts/check-visual-approval.sh)
endif()
